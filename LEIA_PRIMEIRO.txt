   $ php artisan config:clear && composer dump-autoload

3. Testar com SQLite:
   Editar .env: DB_CONNECTION=sqlite
   $ rm database/database.sqlite
   $ php artisan migrate:fresh

4. Se funcionar, testar com MySQL:
   Editar .env: DB_CONNECTION=mysql
   $ php artisan migrate:fresh
